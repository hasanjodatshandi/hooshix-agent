# Add this to your VPS Caddyfile (deploy/gateway/Caddyfile)
# This routes agent.hooshix.com to the Node.js MCP server

agent.hooshix.com {
    tls {
        on_demand
    }

    reverse_proxy 127.0.0.1:18898 {
        health_uri /health
        health_interval 10s
        health_timeout 3s
        transport http {
            response_header_timeout 60s
        }
    }

    log {
        output stdout
        format json
    }
}
